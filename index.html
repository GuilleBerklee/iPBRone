  function init3DViewer() {

    scene = new THREE.Scene();

    var isDragging = false;

    var previousMousePosition = { x: 0, y: 0 };


    function onPointerDown(e) {

    }


    function onPointerMove(e) {

      if (!isDragging) return;

      var clientX = e.clientX || (e.touches && e.touches[0].clientX);

      var clientY = e.clientY || (e.touches && e.touches[0].clientY);

      

      var deltaX = clientX - previousMousePosition.x;

      var deltaY = clientY - previousMousePosition.y;


      mesh.rotation.y += deltaX * 0.008;

      mesh.rotation.x += deltaY * 0.008;


      previousMousePosition = { x: clientX, y: clientY };

    }


    function onPointerUp() { isDragging = false; }


    container.addEventListener('pointerdown', onPointerDown);

    container.addEventListener('pointermove', onPointerMove);

    window.addEventListener('pointerup', onPointerUp);

  function changeMeshGeometry(type) {

    } else if (type === 'plane') {

    }

  }

  // ---------- Camera & Torch ----------

  async function startCamera() {

    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {

      statusEl.textContent = 'Este navegador no soporta acceso a la cámara.';

      return;

    }

    try {

      statusEl.textContent = 'Solicitando acceso a la cámara…';

      stream = await navigator.mediaDevices.getUserMedia({

        video: { facingMode: { ideal: 'environment' }, width: { ideal: 1920 }, height: { ideal: 1920 } },

        audio: false

      });

      video.srcObject = stream;

      await video.play();

      track = stream.getVideoTracks()[0];

      checkTorchSupport();

      placeholder.style.display = 'none';

      shutterBtn.disabled = false;

      statusEl.textContent = 'Cámara lista';

    } catch (err) {

      console.error(err);

      statusEl.textContent = 'No se pudo acceder a la cámara. Revisa los permisos del navegador.';

    }

  }


  function checkTorchSupport() {

    if (!track) return;

    const capabilities = track.getCapabilities ? track.getCapabilities() : {};

    torchBtn.disabled = !capabilities.torch;

  }


  async function toggleTorch() {

    if (!track) return;

    try {

      isTorchOn = !isTorchOn;

      await track.applyConstraints({ advanced: [{ torch: isTorchOn }] });

      torchBtn.classList.toggle('active', isTorchOn);

    } catch (err) {

      isTorchOn = false;

      torchBtn.classList.remove('active');

    }

  }


  function stopCamera() {

    if (stream) {

      if (isTorchOn && track) {

        track.applyConstraints({ advanced: [{ torch: false }] }).catch(() => {});

        isTorchOn = false;

        torchBtn.classList.remove('active');

      }

      stream.getTracks().forEach(t => t.stop());

      stream = null; track = null; torchBtn.disabled = true;

    }

  }


  // ---------- Math helpers ----------

  function clampVal(v){ return v < 0 ? 0 : (v > 255 ? 255 : v); }

  function clampIndex(i, n){ return i < 0 ? 0 : (i >= n ? n - 1 : i); }


  function boxBlurH(src, w, h, radius){

    var out = new Float32Array(w*h);

    var windowSize = radius*2+1;

    for (var y=0; y<h; y++){

      var rowStart = y*w, sum = 0;

      for (var k=-radius; k<=radius; k++) sum += src[rowStart + clampIndex(k, w)];

      for (var x=0; x<w; x++){

        out[rowStart+x] = sum / windowSize;

        sum += src[rowStart+clampIndex(x+radius+1, w)] - src[rowStart+clampIndex(x-radius, w)];

      }

    }

    return out;

  }


  function boxBlurV(src, w, h, radius){

    var out = new Float32Array(w*h);

    var windowSize = radius*2+1;

    for (var x=0; x<w; x++){

      var sum = 0;

      for (var k=-radius; k<=radius; k++) sum += src[clampIndex(k, h)*w + x];

      for (var y=0; y<h; y++){

        out[y*w+x] = sum / windowSize;

        sum += src[clampIndex(y+radius+1, h)*w+x] - src[clampIndex(y-radius, h)*w+x];

      }

    }

    return out;

  }


  function boxBlur(src, w, h, radius){

    if (radius < 1) return src.slice();

    return boxBlurV(boxBlurH(src, w, h, radius), w, h, radius);

  }


  function computeLuminance(imageData){

    var data = imageData.data;

    var L = new Float32Array(imageData.width * imageData.height);

    for (var i=0, p=0; i<data.length; i+=4, p++){

      L[p] = (0.299*data[i] + 0.587*data[i+1] + 0.114*data[i+2]) / 255;

    }

    return L;

  }


  // ---------- Map generators ----------

  function computeAlbedo(imageData, lowFreq, w, h){

    var data = imageData.data, out = new ImageData(w, h), od = out.data, meanLow = 0;

    for (var i=0; i<lowFreq.length; i++) meanLow += lowFreq[i];

    meanLow /= lowFreq.length;

    for (var p=0, j=0; p<w*h; p++, j+=4){

      var factor = Math.min(Math.max(meanLow / Math.max(lowFreq[p], 0.02), 0.35), 2.8);

      od[j]   = clampVal(data[j]   * factor);

      od[j+1] = clampVal(data[j+1] * factor);

      od[j+2] = clampVal(data[j+2] * factor);

      od[j+3] = 255;

    }

    return out;

  }


  function computeNormal(Lsmooth, w, h, strength){

    var out = new ImageData(w, h), od = out.data;

    for (var y=0; y<h; y++){

      for (var x=0; x<w; x++){

        var xm = clampIndex(x-1, w), xp = clampIndex(x+1, w);

        var ym = clampIndex(y-1, h), yp = clampIndex(y+1, h);

        var gx = (Lsmooth[y*w+xp] - Lsmooth[y*w+xm]) * 0.5;

        var gy = (Lsmooth[yp*w+x] - Lsmooth[ym*w+x]) * 0.5;

        var nx = -gx * strength, ny = -gy * strength, nz = 1;

        var len = Math.sqrt(nx*nx + ny*ny + nz*nz) || 1;

        var idx = (y*w+x)*4;

        od[idx]   = clampVal(((nx/len)*0.5+0.5)*255);

        od[idx+1] = clampVal(((ny/len)*0.5+0.5)*255);

        od[idx+2] = clampVal(((nz/len)*0.5+0.5)*255);

        od[idx+3] = 255;

      }

    }

    return out;

  }


  function computeRoughness(L, w, h, invert){

    var radius = 3, L2 = new Float32Array(L.length);

    for (var i=0; i<L.length; i++) L2[i] = L[i]*L[i];

    var meanL = boxBlur(L, w, h, radius), meanL2 = boxBlur(L2, w, h, radius);

    var std = new Float32Array(L.length), maxStd = 1e-5;

    for (i=0; i<L.length; i++){

      var v = Math.max(0, meanL2[i] - meanL[i]*meanL[i]);

      std[i] = Math.sqrt(v);

      if (std[i] > maxStd) maxStd = std[i];

    }

    var out = new ImageData(w, h), od = out.data;

    for (var p=0, j=0; p<w*h; p++, j+=4){

      var val = std[p] / maxStd;

      if (invert) val = 1 - val;

      var g = clampVal(val*255);

      od[j]=g; od[j+1]=g; od[j+2]=g; od[j+3]=255;

    }

    return out;

  }


  function computeAO(L, w, h, strength){

    var local = boxBlur(L, w, h, 4), out = new ImageData(w, h), od = out.data;

    for (var p=0, j=0; p<w*h; p++, j+=4){

      var occlusion = Math.max(0, local[p]-L[p]) * strength;

      var g = clampVal(Math.min(1, Math.max(0, 1-occlusion))*255);

      od[j]=g; od[j+1]=g; od[j+2]=g; od[j+3]=255;

    }

    return out;

  }


  function computeMetallic(w, h, value){

    var out = new ImageData(w, h), od = out.data, g = clampVal(value*255);

    for (var i=0; i<od.length; i+=4){ od[i]=g; od[i+1]=g; od[i+2]=g; od[i+3]=255; }

    return out;

  }


  function drawToCanvas(canvas, imageData){

    canvas.width = imageData.width; canvas.height = imageData.height;

    canvas.getContext('2d').putImageData(imageData, 0, 0);

  }


  // ---------- Pipeline ----------

  function runPipeline(){

    if (!cachedImageData) return;

    var w = cachedW, h = cachedH;

    var Lsmooth = boxBlur(cachedL, w, h, 1);

    var lowFreqBig = boxBlur(cachedL, w, h, Math.max(8, Math.round(Math.min(w, h) * 0.15)));


    drawToCanvas(albedoCanvas, computeAlbedo(cachedImageData, lowFreqBig, w, h));

    drawToCanvas(normalCanvas, computeNormal(Lsmooth, w, h, parseFloat(normalStrengthInput.value)));

    drawToCanvas(roughnessCanvas, computeRoughness(cachedL, w, h, roughnessInvertInput.checked));

    drawToCanvas(aoCanvas, computeAO(cachedL, w, h, parseFloat(aoStrengthInput.value)));

    drawToCanvas(metallicCanvas, computeMetallic(w, h, parseFloat(metallicInput.value)));


    var metaTxt = w + '×' + h + ' · PNG';

    document.getElementById('albedoMeta').textContent = metaTxt;

    document.getElementById('normalMeta').textContent = metaTxt;

    document.getElementById('roughnessMeta').textContent = metaTxt;

    document.getElementById('aoMeta').textContent = metaTxt;


    update3DTextures();

  }


  function capturePhoto(){

    if (!stream) return;

    var vw = video.videoWidth, vh = video.videoHeight;

    if (!vw || !vh) return;


    var cropSize = Math.min(vw, vh);

    var sx = (vw - cropSize) / 2, sy = (vh - cropSize) / 2;

    var outSize = Math.min(1024, cropSize);


    var tmp = document.createElement('canvas');

    tmp.width = outSize; tmp.height = outSize;

    var ctx = tmp.getContext('2d');

    ctx.drawImage(video, sx, sy, cropSize, cropSize, 0, 0, outSize, outSize);


    cachedImageData = ctx.getImageData(0, 0, outSize, outSize);

    cachedW = outSize; cachedH = outSize;

    cachedL = computeLuminance(cachedImageData);


    drawToCanvas(sourceCanvas, cachedImageData);

    sourceMeta.textContent = outSize + '×' + outSize + ' px';


    runPipeline();

    showResults();

  }

  function downloadCanvas(canvas, filename){

    canvas.toBlob(function(blob){

      var url = URL.createObjectURL(blob);

      var a = document.createElement('a');

      a.href = url; a.download = filename;

      document.body.appendChild(a); a.click(); a.remove();

      URL.revokeObjectURL(url);

    }, 'image/png');

  }


  async function downloadAll(){

    var zip = new JSZip();

    var items = [

      [albedoCanvas, 'albedo.png'], [normalCanvas, 'normal.png'],

      [roughnessCanvas, 'roughness.png'], [aoCanvas, 'ao.png'],

      [metallicCanvas, 'metallic.png']

    ];

    for (var i=0; i<items.length; i++){

      var canvas = items[i][0], name = items[i][1];

      var blob = await new Promise(resolve => canvas.toBlob(resolve, 'image/png'));

      zip.file(name, blob);

    }

    var content = await zip.generateAsync({ type: 'blob' });

    var url = URL.createObjectURL(content);

    var a = document.createElement('a');

    a.href = url; a.download = 'material-pbr.zip';

    document.body.appendChild(a); a.click(); a.remove();

    URL.revokeObjectURL(url);

  }
